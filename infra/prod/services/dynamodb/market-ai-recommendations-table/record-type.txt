table_name: trii-prod-market-ai-recommendations

purpose:
- persist automated market recommendations generated from triggered market conditions
- store one recommendation record per triggered bundle of rules
- provide a browser-consumable table for the latest AI-assisted market view

primary_key:
- pk: trigger_signature

global_secondary_indexes:
- gsi1
  pk: symbol
  sk: created_at
  purpose: query recommendations by symbol from newest to oldest

record_type:
- trigger_signature: string, deterministic signature built from symbol, captured_at, and triggered rule bundle
- symbol: string, uppercase stock symbol
- captured_at: string, iso-8601 timestamp with timezone for the triggering snapshot
- snapshot_checksum: string
- current_snapshot_key: map, symbol and captured_at of the triggering snapshot
- previous_snapshot_key: map, symbol and captured_at of the immediately previous snapshot when available
- current_stats_pk: string, formatted as SYMBOL
- triggered_rules: list<string>, full rule bundle that triggered the recommendation
- zscore_context: map, selected z-score, mean, stddev, latest_value, and sample_count details when available
- recommendation_status: string, placeholder, generated, or failed
- recommendation_summary: string
- recommendation_details: map, structured response payload stored for browser consumption
- model_id: string
- model_invoked: boolean
- created_at: string, recommendation creation timestamp

query_patterns:
- latest recommendation for one symbol
- audit recommendation by deterministic trigger signature
