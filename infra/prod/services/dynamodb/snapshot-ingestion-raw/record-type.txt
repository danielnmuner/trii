table_name: trii-prod-snapshot-ingestion-raw

purpose:
- persist the original accepted snapshot payload for a short operational debugging window
- keep a raw trail separate from the longer-lived current-snapshots table
- expire automatically after 24 hours

primary_key:
- pk: symbol
- sk: captured_at

global_secondary_indexes:
- gsi1
  pk: captured_date
  sk: symbol#captured_at
  purpose: inspect accepted raw snapshots by day without scanning

record_type:
- symbol: string, uppercase stock symbol
- asset_name: string
- currency: string
- captured_at: string, iso-8601 timestamp with timezone
- captured_date: string, yyyy-mm-dd
- timezone: string
- snapshot_checksum: string
- symbol_captured_at: string, formatted as SYMBOL#captured_at
- source: string
- source_url: string
- trigger_reason: string
- last_price: number
- daily_change_amount: number
- daily_change_percent: number
- daily_change_direction: string
- previous_close: number
- best_bid_price: number
- best_bid_quantity: number
- best_ask_price: number
- best_ask_quantity: number
- high_price: number
- low_price: number
- traded_value: number
- traded_volume: number
- bid_levels: list<map>
- ask_levels: list<map>
- raw_indicators: map
- extension_meta: map
- expires_at: number, unix epoch seconds used by DynamoDB TTL for 24-hour expiry

query_patterns:
- short-lived raw audit by symbol and timestamp
- short-lived raw audit by capture date
