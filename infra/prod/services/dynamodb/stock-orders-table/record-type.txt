table_name: trii-prod-stock-orders

purpose:
- store historical order records uploaded by the user
- scope every order record to one user
- prevent duplicate inserts across repeated historical uploads for the same user

primary_key:
- pk: record_checksum

global_secondary_indexes:
- gsi1
  pk: user_name#symbol
  sk: created_at
  purpose: query one user's order history by symbol

- gsi2
  pk: user_name#created_month
  sk: created_at#symbol
  purpose: query one user's monthly order history without scanning

record_type:
- user_name: string, normalized user identifier in lowercase and safe for S3 prefixes
- record_checksum: string, globally unique normalized checksum for the user-scoped record
- source_file_checksum: string
- source_line_number: number
- created_at: string, iso-8601 timestamp with timezone in America/Bogota
- created_month: string, yyyy-mm
- user_symbol: string, materialized gsi partition key formatted as user_name#SYMBOL
- user_created_month: string, materialized gsi partition key formatted as user_name#YYYY-MM
- created_at_symbol: string, materialized gsi sort key formatted as created_at#SYMBOL
- symbol: string, uppercase stock symbol
- order_side: string, buy or sell
- raw_status: string
- normalized_status: string
- requested_quantity: number
- filled_quantity: number
- pending_quantity: number
- price_per_share: number
- gross_amount: number
- commission_amount: number
- net_amount: number
- currency: string
- imported_at: string, iso-8601 timestamp with timezone in America/Bogota

duplicate_rule:
- every record_checksum must be unique for the same normalized user
- if any checksum in the uploaded file is duplicated, reject the full upload
- if a checksum for the same user already exists in dynamodb, skip that record and persist only new ones

query_patterns:
- order history by user and symbol
- order history by user and month
- exact lookup by user and checksum
