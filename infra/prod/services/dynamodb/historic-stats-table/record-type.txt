table_name: trii-prod-historic-stats

purpose:
- store the official incremental statistics used by analytics
- avoid scanning raw snapshots when streamlit only needs current value, mu, sigma, and sample size
- support future metric expansion without changing the raw snapshots table
- recompute business metrics from raw order-book fields before persisting aggregates, instead of trusting pre-calculated values from current-snapshots
- persist only the metrics that truly benefit from an incremental market sample and z-score ready state

primary_key:
- pk: symbol
- sk: metric or data_quality#yyyy-mm-dd

global_secondary_indexes:
- gsi1
  pk: symbol#metric
  sk: last_source_captured_at
  purpose: query one metric timeline for one symbol without scanning

official_statistical_metrics_from_raw_snapshot:
- spread_bps = (spread / mid_price) * 10000
- obi_l1 = (best_bid_quantity - best_ask_quantity) / (best_bid_quantity + best_ask_quantity)
- obi_top_5 = (bid_depth_total_5 - ask_depth_total_5) / (bid_depth_total_5 + ask_depth_total_5)
- book_pressure_ratio = bid_depth_total_5 / ask_depth_total_5
- depth_weighted_microprice_deviation = microprice - mid_price
- traded_volume = cumulative traded volume reported in the snapshot
- traded_value = cumulative traded value reported in the snapshot
- volume_rate = (traded_volume_t - traded_volume_t-1) / delta_seconds for contiguous snapshots in the same trading day
- value_rate = (traded_value_t - traded_value_t-1) / delta_seconds for contiguous snapshots in the same trading day

special_backfill_supported_record_types:
- seasonality_profile = rebuild intraday seasonality buckets from traded_volume and traded_value deltas between contiguous snapshots

excluded_operational_snapshot_fields:
- last_price
- daily_change_amount
- daily_change_percent
- spread
- best_bid_price
- best_ask_price
- best_bid_quantity
- best_ask_quantity
- mid_price
- microprice
- bid_depth_total_5
- ask_depth_total_5

query_patterns:
- fetch the full incremental market sample for one symbol
- fetch the current trading-day data quality record for one symbol
- fetch one metric timeline for one symbol through gsi1

backfill_policy:
- historical rebuilds for newly introduced metrics are executed manually through historic-stats-backfill
- current-snapshots remains the deduplicated raw source of truth for any backfill
- live writes stay controlled by local.enabled_statistical_metrics so a new metric can be backfilled before it is enabled in real time
- seasonality_profile is maintained by a dedicated live-updater branch and is not part of local.enabled_statistical_metrics
