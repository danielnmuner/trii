table_name: trii-prod-historic-stats

purpose:
- store the official incremental statistics used by analytics
- avoid scanning raw snapshots when streamlit only needs current value, mu, sigma, and sample size
- support future metric expansion without changing the raw snapshots table
- recompute business metrics from raw order-book fields before persisting aggregates, instead of trusting pre-calculated values from current-snapshots
- persist only the metrics that truly benefit from an incremental market sample and z-score ready state

primary_key:
- pk: symbol
- sk: stats_summary or seasonality_profile

global_secondary_indexes:
- gsi1
  pk: symbol#metric
  sk: last_source_captured_at
  purpose: query one metric timeline for one symbol without scanning

stats_summary_record_type:
- pk: string, uppercase stock symbol
- sk: string, fixed value stats_summary
- symbol: string, uppercase stock symbol
- record_type: string, stats_summary
- metric_count: number, count of embedded metrics currently maintained in metrics
- metrics: map keyed by metric name
- metrics.<metric>.metric: string, one of spread_bps, obi_l1, obi_top_5, traded_volume, traded_value, or vwap
- metrics.<metric>.sample_count: number
- metrics.<metric>.mean: number
- metrics.<metric>.m2: number, Welford running second moment
- metrics.<metric>.stddev: number
- metrics.<metric>.min_value: number
- metrics.<metric>.max_value: number
- metrics.<metric>.latest_value: number
- missing_metrics: list<string>, enabled metrics not present in the current stats_summary item
- last_source_captured_at: string, latest source snapshot timestamp folded into the summary
- last_source_checksum: string, checksum of the latest source snapshot folded into the summary
- last_updated_at: string, iso-8601 timestamp with timezone
- stats_scope: string, currently all_time_market_sample
- stats_version: number, optimistic concurrency counter on the whole summary item

seasonality_profile_record_type:
- pk: string, uppercase stock symbol
- sk: string, seasonality_profile
- symbol: string, uppercase stock symbol
- record_type: string, seasonality_profile
- bucket_granularity_minutes: number, currently 30
- timezone: string, currently America/Bogota
- total_days_processed: number, trading days successfully folded into the profile
- total_snapshots_processed: number, bucket-close writes processed for the symbol
- last_source_captured_at: string, iso-8601 timestamp with timezone
- last_updated_at: string, iso-8601 timestamp with timezone
- stats_scope: string, currently weekly_intraday_seasonality
- stats_version: number, optimistic concurrency counter
- pending_day: optional map used by the live updater until a trading day can be finalized into weekly_profile
- pending_day.trading_date: string, yyyy-mm-dd in America/Bogota
- pending_day.weekday: string, ISO weekday number
- pending_day.last_source_captured_at: string, latest bucket-close source timestamp for the day
- pending_day.total_day_volume: number, cumulative traded volume at the latest processed bucket close
- pending_day.total_day_value: number, cumulative traded value at the latest processed bucket close
- pending_day.last_processed_bucket_key: string | null, last closed bucket already persisted for the day
- pending_day.last_processed_total_volume: number, cumulative traded volume at the last processed bucket close
- pending_day.last_processed_total_value: number, cumulative traded value at the last processed bucket close
- pending_day.hours.<hh:mm>.bucket_volume: number
- pending_day.hours.<hh:mm>.bucket_value: number
- weekly_profile: map keyed by ISO weekday number
- weekly_profile.<weekday>.weekday_label: string, monday through friday for trading sessions
- weekly_profile.<weekday>.days_processed: number
- weekly_profile.<weekday>.accumulated_day_volume: number
- weekly_profile.<weekday>.accumulated_day_value: number
- weekly_profile.<weekday>.hours.<hh:mm>.accumulated_volume: number
- weekly_profile.<weekday>.hours.<hh:mm>.accumulated_value: number
- weekly_profile.<weekday>.hours.<hh:mm>.delta_samples: number, completed 30-minute bucket observations folded into the bucket
- weekly_profile.<weekday>.hours.<hh:mm>.bucket_vwap: number
- weekly_profile.<weekday>.hours.<hh:mm>.volume_share_stats.sample_count: number
- weekly_profile.<weekday>.hours.<hh:mm>.volume_share_stats.mu: number
- weekly_profile.<weekday>.hours.<hh:mm>.volume_share_stats.m2: number
- weekly_profile.<weekday>.hours.<hh:mm>.volume_share_stats.variance: number
- weekly_profile.<weekday>.hours.<hh:mm>.volume_share_stats.sigma: number
- weekly_profile.<weekday>.hours.<hh:mm>.vwap_stats.sample_count: number
- weekly_profile.<weekday>.hours.<hh:mm>.vwap_stats.mu: number
- weekly_profile.<weekday>.hours.<hh:mm>.vwap_stats.m2: number
- weekly_profile.<weekday>.hours.<hh:mm>.vwap_stats.variance: number
- weekly_profile.<weekday>.hours.<hh:mm>.vwap_stats.sigma: number

official_statistical_metrics_from_raw_snapshot:
- spread_bps = (spread / mid_price) * 10000
- obi_l1 = (best_bid_quantity - best_ask_quantity) / (best_bid_quantity + best_ask_quantity)
- obi_top_5 = (bid_depth_total_5 - ask_depth_total_5) / (bid_depth_total_5 + ask_depth_total_5)
- traded_volume = cumulative traded volume reported in the snapshot
- traded_value = cumulative traded value reported in the snapshot
- vwap = traded_value / traded_volume when traded_volume > 0

special_record_types:
- stats_summary = live-updated compact record that embeds the six maintained statistical metrics for a symbol
- seasonality_profile = fold intraday seasonality buckets from cumulative traded_volume and traded_value only when a 30-minute bucket closes

excluded_operational_snapshot_fields:
- last_price
- daily_change_amount
- daily_change_percent
- spread
- best_bid_price
- best_ask_price
- best_bid_quantity
- best_ask_quantity
- mid_price
- microprice
- bid_depth_total_5
- ask_depth_total_5

query_patterns:
- fetch the compact stats_summary record for one symbol
- fetch the seasonality_profile record for one symbol

write_policy:
- current-snapshots remains the deduplicated raw source of truth for live incremental updates
- live writes stay controlled by local.enabled_statistical_metrics and are folded directly into stats_summary
- seasonality_profile is maintained only when a 30-minute bucket closes and is not part of local.enabled_statistical_metrics
