table_name: trii-prod-daily-closing-snapshots

purpose:
- store one end-of-day market snapshot per symbol and trading date
- preserve the last available current-snapshot record observed for each symbol on each day
- let the scheduled Lambda persist the target trading day's close without historical range rebuilds

primary_key:
- pk: symbol
- sk: trading_date

global_secondary_indexes:
- gsi1
  pk: trading_date
  sk: symbol
  purpose: inspect all symbols consolidated for one trading date

record_type:
- symbol: string, uppercase stock symbol
- trading_date: string, yyyy-mm-dd in America/Bogota
- record_type: string, fixed as daily_closing_snapshot
- asset_name: string
- currency: string
- timezone: string, fixed as America/Bogota
- source_captured_at: string, iso-8601 timestamp with timezone in America/Bogota
- source_snapshot_checksum: string
- stored_at: string, iso-8601 timestamp with timezone in America/Bogota
- last_price: number
- daily_change_amount: number
- daily_change_percent: number
- previous_close: number
- best_bid_price: number
- best_ask_price: number
- high_price: number
- low_price: number
- traded_value: number
- traded_volume: number

duplicate_rule:
- only one record is stored per symbol and trading_date
- if a record already exists for that key, the scheduled job skips it

query_patterns:
- daily closing history for one symbol
- all daily closing records for one trading date
