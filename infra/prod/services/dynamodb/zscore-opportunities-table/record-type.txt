table_name: trii-prod-zscore-opportunities

purpose:
- persist realtime z-score opportunity events derived from current snapshots
- capture all monitored z-scores for snapshots where at least one metric crosses the trigger threshold, together with full order-book context and approved FIFO inventory state
- support intraday analytics on symbols and opportunity windows without rescanning current snapshots

primary_key:
- pk: snapshot_checksum

global_secondary_indexes:
- gsi1
  pk: symbol
  sk: captured_at
  purpose: query opportunity events for one symbol chronologically
- gsi2
  pk: trading_date
  sk: symbol_captured_at
  purpose: query all opportunity events for one Bogota trading day

record_type:
- snapshot_checksum: string, unique checksum of the triggering snapshot
- symbol: string, uppercase stock symbol
- captured_at: string, iso-8601 timestamp with timezone in America/Bogota
- trading_date: string, yyyy-mm-dd in America/Bogota
- symbol_captured_at: string, formatted as SYMBOL#captured_at
- created_at: string, iso-8601 timestamp with timezone when the opportunity record was persisted
- triggered_z_scores: map keyed by monitored metric name; persisted when at least one monitored metric crosses the trigger threshold
- triggered_z_scores.<metric>.sample_value: number
- triggered_z_scores.<metric>.z_score: number
- last_price: number | null
- daily_change_amount: number | null
- daily_change_percent: number | null, stored as a fraction (for example 0.0028 for 0.28%)
- previous_close: number | null
- high_price: number | null
- low_price: number | null
- bid_levels: list<map>, full bid ladder from the triggering snapshot
- ask_levels: list<map>, full ask ladder from the triggering snapshot
- approved_position_summary: map derived from approved stock orders with FIFO
- approved_position_summary.symbol: string
- approved_position_summary.approved_buy_quantity: number
- approved_position_summary.approved_sell_quantity: number
- approved_position_summary.available_quantity: number
- approved_position_summary.weighted_average_price: number | null

query_patterns:
- latest triggered z-score events for one symbol
- all triggered opportunities for one Bogota trading day
- exact audit lookup for one snapshot checksum
