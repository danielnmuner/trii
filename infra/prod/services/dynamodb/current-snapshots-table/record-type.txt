table_name: trii-prod-current-snapshots

purpose:
- store the latest parsed market snapshots from the primary indicators contract as a time series per symbol
- support calculations and historical analysis mainly by time

primary_key:
- pk: symbol
- sk: captured_at

global_secondary_indexes:
- gsi1
  pk: captured_date
  sk: symbol#captured_at
  purpose: query all snapshots captured on the same date across symbols

record_type:
- symbol: string, uppercase stock symbol
- asset_name: string
- currency: string
- captured_at: string, iso-8601 timestamp with timezone
- captured_date: string, yyyy-mm-dd
- timezone: string
- snapshot_checksum: string
- symbol_captured_at: string, materialized gsi sort key formatted as SYMBOL#captured_at
- last_price: number
- daily_change_amount: number
- daily_change_percent: number
- daily_change_direction: string
- previous_close: number
- best_bid_price: number
- best_bid_quantity: number
- best_ask_price: number
- best_ask_quantity: number
- spread: number
- mid_price: number
- high_price: number
- low_price: number
- traded_value: number
- traded_volume: number
- bid_levels: list<map>, up to five visible bid depth levels with level, quantity, price
- ask_levels: list<map>, up to five visible ask depth levels with level, quantity, price

query_patterns:
- latest snapshot by symbol
- full time series by symbol
- all snapshots for one capture date
