table_name: trii-prod-current-snapshots

purpose:
- store the latest parsed market snapshots from the primary indicators contract as a short-lived per-symbol buffer
- preserve only the two most recent snapshots per symbol for current/previous comparisons
- act as the raw source of truth for stream-driven analytics updates

primary_key:
- pk: symbol
- sk: captured_at

global_secondary_indexes:
- gsi1
  pk: captured_date
  sk: symbol#captured_at
  purpose: query all snapshots captured on the same date across symbols

record_type:
- symbol: string, uppercase stock symbol
- asset_name: string
- currency: string
- captured_at: string, iso-8601 timestamp with timezone
- captured_date: string, yyyy-mm-dd
- timezone: string
- snapshot_checksum: string
- symbol_captured_at: string, materialized gsi sort key formatted as SYMBOL#captured_at
- source: string
- source_url: string
- trigger_reason: string
- last_price: number
- daily_change_amount: number
- daily_change_percent: number
- daily_change_direction: string
- previous_close: number
- best_bid_price: number
- best_bid_quantity: number
- best_ask_price: number
- best_ask_quantity: number
- spread: number
- spread_bps: number
- mid_price: number
- microprice: number
- depth_weighted_microprice_deviation: number
- high_price: number
- low_price: number
- traded_value: number
- traded_volume: number
- bid_depth_total_5: number
- ask_depth_total_5: number
- book_pressure_ratio: number
- obi_l1: number
- obi_top_5: number
- bid_levels: list<map>, up to five visible bid depth levels with level, quantity, price
- ask_levels: list<map>, up to five visible ask depth levels with level, quantity, price
- raw_indicators: map, raw best bid and best ask header fragments when available
- extension_meta: map, browser extension metadata such as version and source tab url

query_patterns:
- latest snapshot by symbol
- previous snapshot by symbol
- at most two buffered snapshots by symbol
- same-day snapshots near market close
- raw source for stream-driven incremental analytics updates

retention_rule:
- a dedicated stream consumer prunes each symbol so only the two newest records remain
- the same pruner can also be invoked manually with `mode=full-scan` to prune the whole table
